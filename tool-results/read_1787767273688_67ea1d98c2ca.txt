    51→function toPersianNum(n: number): string {
    52→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    53→  return String(n).replace(/\d/g, (d) => persianDigits[parseInt(d)]);
    54→}
