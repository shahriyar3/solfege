    99→    99→// ─── Frequency helper ──────────────────────────────────────
   100→   100→
   101→   101→const A4 = 440.0;
   102→   102→const A4_MIDI = 69;
   103→   103→const NOTE_SEMITONES: Record<string, number> = {
   104→   104→  C: 0, 'C#': 1, D: 2, 'D#': 3, E: 4, F: 5,
   105→   105→  'F#': 6, G: 7, 'G#': 8, A: 9, 'A#': 10, B: 11,
   106→   106→};
   107→   107→
   108→   108→function noteFreq(note: string, octave: number): number {
   109→   109→  const midi = (octave + 1) * 12 + (NOTE_SEMITONES[note] ?? 0);
   110→   110→  return Math.round(A4 * Math.pow(2, (midi - A4_MIDI) / 12) * 10) / 10;
   111→   111→}
   112→   112→

   114→   114→  return { note, solfege, octave, frequency: noteFreq(note, octave), duration };
   115→   115→}
