    28→    28→const NOTE_NAMES = ['C', 'C#', 'D', 'D#', 'E', 'F', 'F#', 'G', 'G#', 'A', 'A#', 'B'];
    29→    29→const SOLFEGE: Record<string, string> = {
    30→    30→  C: 'دو', 'C#': 'دو#', D: 'رِ', 'D#': 'رِ#',
    31→    31→  E: 'می', F: 'فا', 'F#': 'فا#',
    32→    32→  G: 'سل', 'G#': 'سل#', A: 'لا', 'A#': 'لا#', B: 'سی',
    33→    33→};
    34→    34→
    35→    35→const A4_FREQ = 440.0;
    36→    36→const A4_MIDI = 69;
    37→    37→
    38→    38→function midiToFreq(midi: number): number {
    39→    39→  return A4_FREQ * Math.pow(2, (midi - A4_MIDI) / 12);
    40→    40→}
    41→    41→
    42→    42→function midiToNoteInfo(midi: number): NoteInfo {
    43→    43→  const idx = ((midi % 12) + 12) % 12;
    44→    44→  const oct = Math.floor(midi / 12) - 1;
    45→    45→  const name = NOTE_NAMES[idx];
    46→    46→  return { note: name, solfege: SOLFEGE[name] ?? name, octave: oct, frequency: Math.round(midiToFreq(midi) * 100) / 100 };
    47→    47→}

   149→   149→export function IntervalTrainer({ currentPitch, isActive, soundEnabled = true }: IntervalTrainerProps) {

   151→   151→  const [phase, setPhase] = useState<Phase>('setup');
   152→   152→
   153→   153→  // Setup

   155→   155→  const [rootName, setRootName] = useState('C');
   156→   156→  const [catId, setCatId] = useState<string | null>(null);
   157→   157→  const [intIdx, setIntIdx] = useState(0);
   158→   158→
   159→   159→  // Play
   160→   160→  const [curInterval, setCurInterval] = useState<IntervalDef | null>(null);
   161→   161→  const [curCat, setCurCat] = useState<IntervalCat | null>(null);
   162→   162→  const [revealed, setRevealed] = useState(false);
   163→   163→  const [playingRoot, setPlayingRoot] = useState(false);
   164→   164→  const [playingTarget, setPlayingTarget] = useState(false);
   165→   165→  const [result, setResult] = useState<'correct' | 'wrong' | null>(null);
   166→   166→  const [resultCents, setResultCents] = useState<number | null>(null);
   167→   167→
   168→   168→  // Score
   169→   169→  const [score, setScore] = useState(0);
   170→   170→  const [attempts, setAttempts] = useState(0);
   171→   171→  const [streak, setStreak] = useState(0);
   172→   172→  const [best, setBest] = useState(0);

   174→   174→  const detected = useRef(false);
   175→   175→  const phaseRef = useRef<Phase>('setup');
   176→   176→  useEffect(() => { phaseRef.current = phase; }, [phase]);
   177→   177→
   178→   178→  const notes = useMemo(() => getNaturalNotes(rootOct), [rootOct]);
   179→   179→  const root = notes.find((n) => n.note === rootName) ?? notes[0];
   180→   180→  const rMidi = rootMidi(root.note, root.octave);
   181→   181→
   182→   182→  const tMidi = useMemo(() => (curInterval ? rMidi + curInterval.semitones : null), [rMidi, curInterval]);
   183→   183→  const target = useMemo(() => (tMidi !== null ? midiToNoteInfo(tMidi) : null), [tMidi]);
   184→   184→
   185→   185→  const avail = useMemo(() => {
   186→   186→    if (!catId) return ALL_INTERVALS;
   187→   187→    const c = CATEGORIES.find((x) => x.id === catId);
   188→   188→    return c ? c.intervals : ALL_INTERVALS;
   189→   189→  }, [catId]);
   190→   190→
   191→   191→  const selInterval = avail[intIdx] ?? avail[0];
   192→   192→
   193→   193→  // Pitch detection
   194→   194→  useEffect(() => {
   195→   195→    if (!isActive || !currentPitch || phaseRef.current !== 'playing' || detected.current || tMidi === null) return;
   196→   196→    const tName = NOTE_NAMES[((tMidi % 12) + 12) % 12];
   197→   197→    const tOct = Math.floor(tMidi / 12) - 1;
   198→   198→    if (currentPitch.note === tName && currentPitch.octave === tOct && Math.abs(currentPitch.cents) <= 20) {
   199→   199→      detected.current = true;

   202→   202→        setResult('correct');
   203→   203→        setResultCents(currentPitch.cents);
   204→   204→        setScore((s) => s + 1);
   205→   205→        setAttempts((a) => a + 1);
   206→   206→        setStreak((s) => { const n = s + 1; setBest((b) => Math.max(b, n)); return n; });
   207→   207→        setRevealed(true);
   208→   208→        setPhase('result');
   209→   209→      });
   210→   210→    }
   211→   211→  }, [currentPitch, isActive, tMidi]);

   214→   214→    const handle = playNote(freq, 1.2);
   215→   215→    setPlayingRoot(true);
   216→   216→    setTimeout(() => { setPlayingRoot(false); handle.stop(); }, 1300);
   217→   217→  }, []);

   222→   222→    if (catId) {
   223→   223→      category = CATEGORIES.find((c) => c.id === catId) ?? null;
   224→   224→      interval = avail[intIdx] ?? avail[0];
   225→   225→    } else {
   226→   226→      interval = ALL_INTERVALS[Math.floor(Math.random() * ALL_INTERVALS.length)];
   227→   227→      category = CATEGORIES.find((c) => c.intervals.some((i) => i.semitones === interval.semitones)) ?? null;
   228→   228→    }
   229→   229→    setCurInterval(interval);
   230→   230→    setCurCat(category);
   231→   231→    setRevealed(false);
   232→   232→    setResult(null);
   233→   233→    setResultCents(null);
   234→   234→    detected.current = false;

   237→   237→  }, [catId, avail, intIdx, root.frequency, playRootNote]);
   238→   238→
   239→   239→  const reveal = useCallback(() => {
   240→   240→    if (soundEnabled) playWrongSound();
   241→   241→    setRevealed(true);
   242→   242→    setResult('wrong');
   243→   243→    setAttempts((a) => a + 1);
   244→   244→    setStreak(0);
   245→   245→    setPhase('result');
   246→   246→  }, [soundEnabled]);
   247→   247→
   248→   248→  const next = useCallback(() => {
   249→   249→    let interval: IntervalDef;
   250→   250→    let category: IntervalCat | null = null;
   251→   251→    if (catId) {
   252→   252→      const nextIdx = (intIdx + 1) % avail.length;
   253→   253→      setIntIdx(nextIdx);
   254→   254→      category = CATEGORIES.find((c) => c.id === catId) ?? null;
   255→   255→      interval = avail[nextIdx];
   256→   256→    } else {
   257→   257→      interval = ALL_INTERVALS[Math.floor(Math.random() * ALL_INTERVALS.length)];
   258→   258→      category = CATEGORIES.find((c) => c.intervals.some((i) => i.semitones === interval.semitones)) ?? null;
   259→   259→    }
   260→   260→    setCurInterval(interval);
   261→   261→    setCurCat(category);
   262→   262→    setRevealed(false);
   263→   263→    setResult(null);
   264→   264→    setResultCents(null);
   265→   265→    detected.current = false;
   266→   266→    setPhase('playing');
   267→   267→    setTimeout(() => playRootNote(root.frequency), 300);
   268→   268→  }, [catId, intIdx, avail, root.frequency, playRootNote]);

   270→   270→  const reset = useCallback(() => {
   271→   271→    setPhase('setup');
   272→   272→    setCurInterval(null);
   273→   273→    setCurCat(null);
   274→   274→    setRevealed(false);
   275→   275→    setResult(null);
   276→   276→    setResultCents(null);
   277→   277→    detected.current = false;
   278→   278→    setScore(0);
   279→   279→    setAttempts(0);
   280→   280→    setStreak(0);
   281→   281→    setBest(0);
   282→   282→  }, []);
   283→   283→
   284→   284→  const playTargetNote = useCallback(() => {
   285→   285→    if (!target) return;
   286→   286→    const handle = playNote(target.frequency, 1.2);
   287→   287→    setPlayingTarget(true);
   288→   288→    setTimeout(() => { setPlayingTarget(false); handle.stop(); }, 1300);
   289→   289→  }, [target]);
   290→   290→
   291→   291→  // ------------------------------------------------------------------
   292→   292→  // Teaser

   295→   295→    return (

   297→   297→        className="border-dashed border-2 border-border/40 bg-gradient-to-br from-muted/10 to-muted/5 hover:border-violet-300/50 transition-all duration-300 group cursor-pointer"

   300→   300→        <CardContent className="p-5 flex items-center gap-4">
   301→   301→          <div className="h-12 w-12 rounded-2xl bg-gradient-to-br from-violet-500 to-indigo-500 flex items-center justify-center shadow-lg shadow-violet-500/20 group-hover:shadow-violet-500/40 transition-shadow">
   302→   302→            <Music className="h-6 w-6 text-white" />

   304→   304→          <div className="flex-1">
   305→   305→            <h3 className="text-sm font-bold">تمرین فاصله</h3>
   306→   306→            <p className="text-xs text-muted-foreground mt-0.5">فاصله‌ها را بشنوید و بخوانید</p>
   307→   307→          </div>
   308→   308→          <ChevronLeft className="h-5 w-5 text-muted-foreground group-hover:text-violet-500 transition-colors" />
