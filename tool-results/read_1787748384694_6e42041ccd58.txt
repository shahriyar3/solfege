    64→    64→function toPersianNum(n: number): string {
    65→    65→  return String(n)

    70→    70→
    71→    71→const A4_FREQ = 440.0;
    72→    72→const A4_MIDI = 69;
    73→    73→
    74→    74→function calcFrequency(note: NoteName, octave: number): number {
    75→    75→  const midi = (octave + 1) * 12 + NOTE_INDICES[note];
    76→    76→  return Math.round(A4_FREQ * Math.pow(2, (midi - A4_MIDI) / 12) * 10) / 10;
    77→    77→}

    82→    82→
    83→    83→interface DetectedNote {
    84→    84→  frequency: number;
    85→    85→  note: string;
    86→    86→  octave: number;
    87→    87→  cents: number;
    88→    88→}
    89→    89→
    90→    90→function autoCorrelate(
    91→    91→  buf: Float32Array,
    92→    92→  sampleRate: number,
    93→    93→  minFreq: number,
    94→    94→  maxFreq: number,
    95→    95→): number | null {
    96→    96→  const SIZE = buf.length;
    97→    97→  let rms = 0;
    98→    98→  for (let i = 0; i < SIZE; i++) {
    99→    99→    rms += buf[i] * buf[i];
   100→   100→  }
   101→   101→  rms = Math.sqrt(rms / SIZE);
   102→   102→  if (rms < 0.01) return null;
   103→   103→
   104→   104→  const minPeriod = Math.floor(sampleRate / maxFreq);
   105→   105→  const maxPeriod = Math.ceil(sampleRate / minFreq);
   106→   106→
   107→   107→  let bestOffset = -1;
   108→   108→  let bestCorrelation = 0;
   109→   109→  let foundGoodCorrelation = false;
   110→   110→  const correlations = new Float32Array(maxPeriod + 1);
   111→   111→
   112→   112→  for (let offset = minPeriod; offset <= Math.min(maxPeriod, Math.floor(SIZE / 2)); offset++) {
   113→   113→    let correlation = 0;
   114→   114→    for (let i = 0; i < SIZE - offset; i++) {
   115→   115→      correlation += Math.abs(buf[i] - buf[i + offset]);
   116→   116→    }
   117→   117→    correlation = 1 - correlation / (SIZE - offset);
   118→   118→    correlations[offset] = correlation;
   119→   119→
   120→   120→    if (correlation > 0.9 && correlation > bestCorrelation) {
   121→   121→      bestCorrelation = correlation;
   122→   122→      bestOffset = offset;
   123→   123→      foundGoodCorrelation = true;
   124→   124→    } else if (foundGoodCorrelation) {
   125→   125→      const shortAvg = (correlations[bestOffset - 1] + correlations[bestOffset] + correlations[bestOffset + 1]) / 3;
   126→   126→      if (correlation < shortAvg) break;
   127→   127→    }
   128→   128→  }
   129→   129→
   130→   130→  if (bestCorrelation > 0.01 && bestOffset > 0) {
   131→   131→    let shift = 0;
   132→   132→    if (bestOffset > 0 && bestOffset < SIZE - 1) {
   133→   133→      const prev = correlations[bestOffset - 1] || 0;
   134→   134→      const next = correlations[bestOffset + 1] || 0;
   135→   135→      shift = (next - prev) / (2 * (2 * bestCorrelation - prev - next));
   136→   136→      if (isNaN(shift)) shift = 0;
   137→   137→    }
   138→   138→    return sampleRate / (bestOffset + shift);
   139→   139→  }
   140→   140→  return null;
   141→   141→}
   142→   142→
   143→   143→function frequencyToNote(freq: number): DetectedNote {
   144→   144→  const midiNumber = Math.round(12 * Math.log2(freq / A4_FREQ) + A4_MIDI);
   145→   145→  const exactMidi = 12 * Math.log2(freq / A4_FREQ) + A4_MIDI;
   146→   146→  const cents = Math.round((exactMidi - midiNumber) * 100);
   147→   147→  const clampedCents = Math.max(-50, Math.min(50, cents));
   148→   148→  const noteIndex = ((midiNumber % 12) + 12) % 12;
   149→   149→  const octave = Math.floor(midiNumber / 12) - 1;
   150→   150→  return {
   151→   151→    frequency: Math.round(freq * 10) / 10,
   152→   152→    note: ALL_NOTE_NAMES[noteIndex],
   153→   153→    octave,
   154→   154→    cents: clampedCents,
   155→   155→  };
   156→   156→}
