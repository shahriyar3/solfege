    46→function toPersianNum(n: number): string {
    47→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    48→  return String(n).replace(/\d/g, (d) => persianDigits[parseInt(d)]);
    49→}
