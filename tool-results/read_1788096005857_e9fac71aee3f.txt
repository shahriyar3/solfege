    99→// ─── Frequency helper ──────────────────────────────────────
   100→
   101→const A4 = 440.0;
   102→const A4_MIDI = 69;
   103→const NOTE_SEMITONES: Record<string, number> = {
   104→  C: 0, 'C#': 1, D: 2, 'D#': 3, E: 4, F: 5,
   105→  'F#': 6, G: 7, 'G#': 8, A: 9, 'A#': 10, B: 11,
   106→};
   107→
   108→function noteFreq(note: string, octave: number): number {
   109→  const midi = (octave + 1) * 12 + (NOTE_SEMITONES[note] ?? 0);
   110→  return Math.round(A4 * Math.pow(2, (midi - A4_MIDI) / 12) * 10) / 10;
   111→}
   112→
   113→function makeStep(note: string, solfege: string, octave: number, duration: number): StepNote {
   114→  return { note, solfege, octave, frequency: noteFreq(note, octave), duration };
   115→}

   377→          steps.push(makeStep(s4[j].note, s4[j].solfege, s4[j].octave, 0.7));
   378→          steps.push(makeStep(center.note, center.solfege, center.octave, 0.7));
   379→        }
   380→        return steps;
   381→      },
   382→    },
   383→  ];
   384→}

   388→export function WarmupModule({ soundEnabled = true }: WarmupModuleProps) {
   389→  const [expanded, setExpanded] = useState(false);
   390→  const [activeExercise, setActiveExercise] = useState<ExerciseId>('ascending');
   391→  const [currentStep, setCurrentStep] = useState(-1);
   392→  const [isPlaying, setIsPlaying] = useState(false);
   393→  const [speed, setSpeed] = useState<SpeedId>('medium');
   394→  const [showDescription, setShowDescription] = useState(true);
   395→
   396→  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   397→  const stopRef = useRef<(() => void) | null>(null);
   398→  const playStepRef = useRef<(idx: number) => void>(() => {});
   399→  const scale = useMemo(() => getSolfeggioScale(4), []);
   400→  const exercises = useMemo(() => getExerciseDefs(), []);
   401→
   402→  const steps = useMemo(
   403→    () => exercises.find((e) => e.id === activeExercise)?.getSteps(scale) ?? [],
   404→    [activeExercise, exercises, scale],
   405→  );
   406→
   407→  const currentExerciseDef = exercises.find((e) => e.id === activeExercise)!;
   408→
   409→  // Cleanup on unmount
   410→  useEffect(() => {
   411→    return () => {
   412→      if (timerRef.current) clearTimeout(timerRef.current);
   413→      if (stopRef.current) stopRef.current();
   414→    };
   415→  }, []);
   416→
   417→  const stopAll = useCallback(() => {
   418→    if (timerRef.current) {
   419→      clearTimeout(timerRef.current);
   420→      timerRef.current = null;
   421→    }
   422→    if (stopRef.current) {
   423→      stopRef.current();
   424→      stopRef.current = null;
   425→    }
   426→    setIsPlaying(false);
   427→    setCurrentStep(-1);
   428→  }, []);
   429→
   430→  // Play a single step: update note name INSTANTLY, then play sound after a tiny delay
   431→  // so the visual update (note name) appears before the audio starts
   432→  const playStepFn = useCallback(
   433→    (stepIndex: number) => {
   434→      if (stepIndex >= steps.length) {
   435→        stopAll();
   436→        return;
   437→      }
   438→
   439→      const step = steps[stepIndex];
   440→      // Update note display IMMEDIATELY — before any work
   441→      setCurrentStep(stepIndex);
   442→
   443→      if (soundEnabled) {
   444→        if (stopRef.current) {
   445→          stopRef.current();
   446→          stopRef.current = null;
   447→        }
   448→        // Small delay so React re-renders the note name BEFORE sound starts
   449→        timerRef.current = setTimeout(() => {
   450→          const handle = playNote(step.frequency, step.duration);
   451→          stopRef.current = handle.stop;
   452→
   453→          // Determine delay before next step
   454→          let delay: number;
   455→          if (currentExerciseDef.hasSpeedControl) {
   456→            delay = SPEED_CONFIG[speed].ms;
   457→          } else if (activeExercise === 'sustain') {
   458→            delay = step.duration * 1000 + 400;
   459→          } else {
   460→            delay = step.duration * 1000 + 200;
   461→          }
   462→
   463→          // Subtract the 50ms we already waited
   464→          const remainingDelay = Math.max(50, delay - 50);
   465→
   466→          timerRef.current = setTimeout(() => {
   467→            playStepRef.current(stepIndex + 1);
   468→          }, remainingDelay);
   469→        }, 50);
   470→      } else {
   471→        // No sound — just advance by the visual delay
   472→        let delay: number;
   473→        if (currentExerciseDef.hasSpeedControl) {
   474→          delay = SPEED_CONFIG[speed].ms;
   475→        } else if (activeExercise === 'sustain') {
   476→          delay = step.duration * 1000 + 400;
   477→        } else {
   478→          delay = step.duration * 1000 + 200;
   479→        }
   480→
   481→        timerRef.current = setTimeout(() => {
   482→          playStepRef.current(stepIndex + 1);
   483→        }, delay);
   484→      }
   485→    },
   486→    [steps, soundEnabled, activeExercise, speed, stopAll, currentExerciseDef.hasSpeedControl],
   487→  );
   488→
   489→  useEffect(() => {
   490→    playStepRef.current = playStepFn;
   491→  }, [playStepFn]);
   492→
   493→  const handlePlay = useCallback(async () => {
   494→    if (isPlaying) {
   495→      stopAll();
   496→      return;
   497→    }
   498→    try { await ensureAudioResumed(); } catch { /* */ }
   499→    setIsPlaying(true);
   500→    setCurrentStep(0);
   501→    playStepRef.current(0);
   502→  }, [isPlaying, stopAll]);
   503→
   504→  const handleExerciseChange = useCallback(
   505→    (id: ExerciseId) => {
   506→      stopAll();
   507→      setActiveExercise(id);
   508→      setShowDescription(true);
   509→    },
   510→    [stopAll],
   511→  );
   512→
   513→  // ─── Teaser ─────────────────────────────────────────────
   514→  if (!expanded) {
   515→    return (
   516→      <Card
   517→        className="border-dashed border-2 border-border/30 bg-gradient-to-br from-muted/10 to-muted/5 hover:border-orange-300/40 transition-all duration-300 group cursor-pointer card-hover"
   518→        onClick={() => setExpanded(true)}
   519→      >
   520→        <CardContent className="p-5 flex items-center gap-4">
   521→          <div className="h-12 w-12 rounded-2xl bg-gradient-to-br from-orange-400 to-red-500 flex items-center justify-center shadow-lg shadow-orange-500/20 group-hover:shadow-orange-500/35 transition-all duration-300 group-hover:scale-105">
   522→            <Flame className="h-6 w-6 text-white" />
   523→          </div>
   524→          <div className="flex-1">
   525→            <h3 className="text-sm font-bold">گرم کردن صدا</h3>
   526→            <p className="text-xs text-muted-foreground mt-0.5">
   527→              تمرینات گرم کردن قبل از سلفژ
   528→            </p>
   529→            <div className="flex gap-1.5 mt-1.5">
   530→              <Badge variant="outline" className="text-[9px] h-4">
   531→                {toPersianNum(exercises.length)} تمرین
   532→              </Badge>
   533→            </div>
   534→          </div>
   535→          <ChevronLeft className="h-5 w-5 text-muted-foreground/50 group-hover:text-orange-500 transition-colors" />
   536→        </CardContent>
   537→      </Card>
   538→    );
   539→  }
   540→
   541→  // ─── Active State ───────────────────────────────────────
   542→  const currentStepData = currentStep >= 0 && currentStep < steps.length
   543→    ? steps[currentStep]
   544→    : null;
   545→
   546→  return (
   547→    <Card className="border border-border/40 bg-card/90 backdrop-blur-sm shadow-lg shadow-black/[0.04] overflow-hidden">
   548→      <CardHeader className="pb-2 pt-4 px-4">
   549→        <div className="flex items-center justify-between">
   550→          <CardTitle className="text-sm font-bold flex items-center gap-2">
   551→            <Flame className="h-4 w-4 text-orange-500" />
   552→            گرم کردن صدا
   553→          </CardTitle>
   554→          <Button
   555→            variant="ghost"
   556→            size="sm"
   557→            onClick={() => {
   558→              stopAll();
   559→              setExpanded(false);
   560→            }}
   561→            className="h-7 w-7 p-0 text-muted-foreground"
   562→            title="بستن"
   563→          >
   564→            <ChevronLeft className="h-4 w-4" />
   565→          </Button>
   566→        </div>
   567→      </CardHeader>
   568→
   569→      <CardContent className="px-4 pb-4 space-y-4">
   570→        {/* Exercise selector — scrollable row */}
   571→        <div className="flex gap-1.5 overflow-x-auto pb-1 -mx-1 px-1" style={{ scrollbarWidth: 'none' }}>
   572→          {exercises.map((ex) => (
   573→            <button
   574→              key={ex.id}
   575→              onClick={() => handleExerciseChange(ex.id)}
   576→              className={cn(
   577→                'flex-shrink-0 flex items-center justify-center gap-1 h-8 rounded-lg text-[11px] font-medium transition-all duration-200 px-2.5',
   578→                activeExercise === ex.id
   579→                  ? 'bg-gradient-to-r from-orange-500 to-red-500 text-white shadow-md shadow-orange-500/25'
   580→                  : 'bg-muted/50 text-muted-foreground hover:bg-muted',
   581→              )}
   582→            >
   583→              {ex.icon}
   584→              <span>{ex.name}</span>
   585→            </button>
   586→          ))}
   587→        </div>
   588→
   589→        {/* Exercise description (collapsible) */}
   590→        <div className="overflow-hidden">
   591→          <button
   592→            onClick={() => setShowDescription((p) => !p)}
   593→            className="flex items-center gap-1.5 text-[11px] text-muted-foreground hover:text-foreground transition-colors w-full"
   594→          >
   595→            <Info className="h-3.5 w-3.5" />
   596→            <span className="font-medium">راهنمای تمرین</span>
   597→            {showDescription ? <ChevronUp className="h-3 w-3 mr-auto" /> : <ChevronDown className="h-3 w-3 mr-auto" />}
   598→          </button>
   599→          <AnimatePresence>
   600→            {showDescription && (
   601→              <motion.div
   602→                initial={{ height: 0, opacity: 0 }}
   603→                animate={{ height: 'auto', opacity: 1 }}
   604→                exit={{ height: 0, opacity: 0 }}
   605→                transition={{ duration: 0.2 }}
   606→                className="overflow-hidden"
   607→              >
   608→                <p className="text-[11px] leading-relaxed text-muted-foreground pt-2 pr-1">
   609→                  {currentExerciseDef.description}
   610→                </p>
   611→              </motion.div>
   612→            )}
   613→          </AnimatePresence>
   614→        </div>
   615→
   616→        {/* Speed control for exercises that support it */}
   617→        {currentExerciseDef.hasSpeedControl && (
   618→          <div className="flex items-center justify-center gap-2">
   619→            <Gauge className="h-3.5 w-3.5 text-muted-foreground" />
   620→            <span className="text-[11px] text-muted-foreground">سرعت:</span>
   621→            {(Object.keys(SPEED_CONFIG) as SpeedId[]).map((s) => (
   622→              <button
   623→                key={s}
   624→                onClick={() => setSpeed(s)}
   625→                className={cn(
   626→                  'h-7 px-2.5 rounded-md text-[11px] font-medium transition-all',
   627→                  speed === s
   628→                    ? 'bg-primary text-primary-foreground shadow-sm'
   629→                    : 'bg-muted/50 text-muted-foreground hover:bg-muted',
   630→                )}
   631→              >
   632→                {SPEED_CONFIG[s].label}
   633→              </button>
   634→            ))}
   635→          </div>
   636→        )}
   637→
   638→        {/* Current note display — circle stays, only text changes */}
   639→        <div className="flex justify-center py-2">
   640→          <div
   641→            className={cn(
   642→              'relative h-32 w-32 rounded-full flex flex-col items-center justify-center transition-all duration-150',
   643→              'bg-gradient-to-br from-muted/80 to-muted/40 border-2',
   644→              currentStepData && isPlaying
   645→                ? 'border-orange-400 shadow-lg shadow-orange-500/20'
   646→                : 'border-border/30 hover:border-border/50',
   647→            )}
   648→          >
   649→            {/* Pulsing ring when playing */}
   650→            {currentStepData && isPlaying && (
   651→              <motion.div
   652→                className="absolute inset-0 rounded-full border-2 border-orange-400/40"
   653→                animate={{ scale: [1, 1.15, 1], opacity: [0.4, 0, 0.4] }}
   654→                transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   655→              />
   656→            )}
   657→
   658→            {currentStepData && isPlaying ? (
   659→              <>
   660→                {/* Note name — instant swap, no exit/enter animation */}
   661→                <span
   662→                  key={`n-${currentStep}`}
   663→                  className={cn(
   664→                    'text-3xl font-bold bg-gradient-to-br bg-clip-text text-transparent',
   665→                    NOTE_GRADIENTS[currentStepData.note] ?? 'from-gray-400 to-gray-500',
   666→                  )}
   667→                >
   668→                  {currentStepData.solfege}
   669→                </span>
   670→                <span
   671→                  key={`s-${currentStep}`}
   672→                  className="text-xs text-muted-foreground mt-0.5"
   673→                >
   674→                  {currentStepData.note}{toPersianNum(currentStepData.octave)}
   675→                </span>
   676→              </>
   677→            ) : (
   678→              <>
   679→                <span className="text-2xl text-muted-foreground/40">
   680→                  {currentExerciseDef.icon}
   681→                </span>
   682→                <span className="text-[11px] text-muted-foreground mt-1">
   683→                  {currentExerciseDef.name}
   684→                </span>
   685→              </>
   686→            )}
   687→          </div>
   688→        </div>
   689→
   690→        {/* Step progress dots */}
   691→        <div className={cn(
   692→          'flex items-center justify-center flex-wrap',
   693→          steps.length > 15 ? 'gap-1' : 'gap-1.5',
   694→        )}>
   695→          {steps.map((step, i) => {
   696→            const isCurrent = i === currentStep && isPlaying;
   697→            const isPast = i < currentStep && isPlaying;
   698→            return (
   699→              <motion.div
   700→                key={i}
   701→                initial={false}
   702→                animate={{
   703→                  scale: isCurrent ? 1.4 : 1,
   704→                  opacity: isPast ? 0.4 : isCurrent ? 1 : 0.25,
   705→                }}
   706→                transition={{ type: 'spring', stiffness: 400, damping: 25 }}
   707→                className={cn(
   708→                  'rounded-full transition-colors duration-200',
   709→                  steps.length > 15 ? 'h-1.5 w-1.5' : 'h-2 w-2',
   710→                  isCurrent
   711→                    ? cn('bg-gradient-to-br', NOTE_GRADIENTS[step.note] ?? 'from-gray-400 to-gray-500')
   712→                    : isPast
   713→                      ? 'bg-muted-foreground/40'
   714→                      : 'bg-border',
   715→                )}
   716→              />
   717→            );
   718→          })}
   719→        </div>
   720→
   721→        {/* Play / Stop button */}
   722→        <div className="flex justify-center">
   723→          <Button
   724→            onClick={handlePlay}
   725→            className={cn(
   726→              'h-11 px-8 rounded-full font-medium transition-all duration-300',
   727→              isPlaying
   728→                ? 'bg-gradient-to-r from-red-500 to-rose-600 hover:from-red-600 hover:to-rose-700 shadow-lg shadow-red-500/25'
   729→                : 'bg-gradient-to-r from-orange-500 to-red-500 hover:from-orange-600 hover:to-red-600 shadow-lg shadow-orange-500/25',
   730→            )}
   731→          >
   732→            <AnimatePresence mode="wait" initial={false}>
   733→              {isPlaying ? (
   734→                <motion.span
   735→                  key="stop"
   736→                  initial={{ scale: 0, rotate: -90 }}

   738→                  exit={{ scale: 0, rotate: 90 }}
   739→                  className="flex items-center gap-2"

   741→                  <Square className="h-4 w-4" fill="currentColor" />
   742→                  توقف
   743→                </motion.span>
   744→              ) : (
   745→                <motion.span

   773→}
