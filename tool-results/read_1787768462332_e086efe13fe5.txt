    99→// ─── Frequency helper ──────────────────────────────────────

   101→const A4 = 440.0;
   102→const A4_MIDI = 69;
   103→const NOTE_SEMITONES: Record<string, number> = {
   104→  C: 0, 'C#': 1, D: 2, 'D#': 3, E: 4, F: 5,
   105→  'F#': 6, G: 7, 'G#': 8, A: 9, 'A#': 10, B: 11,
   106→};
   107→
   108→function noteFreq(note: string, octave: number): number {
   109→  const midi = (octave + 1) * 12 + (NOTE_SEMITONES[note] ?? 0);
   110→  return Math.round(A4 * Math.pow(2, (midi - A4_MIDI) / 12) * 10) / 10;
   111→}

   428→  }, []);
   429→
   430→  // Play a single step: update note name INSTANTLY, then play sound
   431→  const playStepFn = useCallback(
   432→    (stepIndex: number) => {
   433→      if (stepIndex >= steps.length) {
   434→        stopAll();
   435→        return;
   436→      }
   437→
   438→      const step = steps[stepIndex];
   439→      // Update note display IMMEDIATELY — before any work
   440→      setCurrentStep(stepIndex);
   441→
   442→      if (soundEnabled) {
   443→        if (stopRef.current) {
   444→          stopRef.current();
   445→          stopRef.current = null;
   446→        }
   447→        const handle = playNote(step.frequency, step.duration);
   448→        stopRef.current = handle.stop;
   449→      }
   450→
   451→      // Determine delay before next step
   452→      let delay: number;
   453→      if (currentExerciseDef.hasSpeedControl) {
   454→        delay = SPEED_CONFIG[speed].ms;
   455→      } else if (activeExercise === 'sustain') {
   456→        delay = step.duration * 1000 + 400;
   457→      } else {
   458→        delay = step.duration * 1000 + 200;
   459→      }
   460→
   461→      timerRef.current = setTimeout(() => {
   462→        playStepRef.current(stepIndex + 1);
   463→      }, delay);
   464→    },
   465→    [steps, soundEnabled, activeExercise, speed, stopAll, currentExerciseDef.hasSpeedControl],
   466→  );
   467→
   468→  useEffect(() => {
   469→    playStepRef.current = playStepFn;
   470→  }, [playStepFn]);
   471→
   472→  const handlePlay = useCallback(() => {
   473→    if (isPlaying) {
   474→      stopAll();
   475→      return;
   476→    }
   477→    ensureAudioReady();
   478→    setIsPlaying(true);
   479→    setCurrentStep(0);
   480→    playStepRef.current(0);
   481→  }, [isPlaying, stopAll]);
   482→
   483→  const handleExerciseChange = useCallback(
   484→    (id: ExerciseId) => {
   485→      stopAll();
   486→      setActiveExercise(id);
   487→      setShowDescription(true);
   488→    },
   489→    [stopAll],
   490→  );
   491→
   492→  // ─── Teaser ─────────────────────────────────────────────
   493→  if (!expanded) {
   494→    return (
   495→      <Card
   496→        className="border-dashed border-2 border-border/30 bg-gradient-to-br from-muted/10 to-muted/5 hover:border-orange-300/40 transition-all duration-300 group cursor-pointer card-hover"
   497→        onClick={() => setExpanded(true)}
   498→      >
   499→        <CardContent className="p-5 flex items-center gap-4">
   500→          <div className="h-12 w-12 rounded-2xl bg-gradient-to-br from-orange-400 to-red-500 flex items-center justify-center shadow-lg shadow-orange-500/20 group-hover:shadow-orange-500/35 transition-all duration-300 group-hover:scale-105">
   501→            <Flame className="h-6 w-6 text-white" />
   502→          </div>
   503→          <div className="flex-1">
   504→            <h3 className="text-sm font-bold">گرم کردن صدا</h3>
   505→            <p className="text-xs text-muted-foreground mt-0.5">
   506→              تمرینات گرم کردن قبل از سلفژ
   507→            </p>
   508→            <div className="flex gap-1.5 mt-1.5">
   509→              <Badge variant="outline" className="text-[9px] h-4">
   510→                {toPersianNum(exercises.length)} تمرین
   511→              </Badge>
   512→            </div>
   513→          </div>
   514→          <ChevronLeft className="h-5 w-5 text-muted-foreground/50 group-hover:text-orange-500 transition-colors" />
   515→        </CardContent>
   516→      </Card>
   517→    );
   518→  }
   519→
   520→  // ─── Active State ───────────────────────────────────────
   521→  const currentStepData = currentStep >= 0 && currentStep < steps.length
   522→    ? steps[currentStep]
   523→    : null;
   524→
   525→  return (
   526→    <Card className="border border-border/40 bg-card/90 backdrop-blur-sm shadow-lg shadow-black/[0.04] overflow-hidden">
   527→      <CardHeader className="pb-2 pt-4 px-4">
   528→        <div className="flex items-center justify-between">
   529→          <CardTitle className="text-sm font-bold flex items-center gap-2">
   530→            <Flame className="h-4 w-4 text-orange-500" />
   531→            گرم کردن صدا
   532→          </CardTitle>
   533→          <Button
   534→            variant="ghost"
   535→            size="sm"
   536→            onClick={() => {
   537→              stopAll();
   538→              setExpanded(false);
   539→            }}
   540→            className="h-7 w-7 p-0 text-muted-foreground"
   541→            title="بستن"
   542→          >
   543→            <ChevronLeft className="h-4 w-4" />
   544→          </Button>
   545→        </div>
   546→      </CardHeader>
   547→
   548→      <CardContent className="px-4 pb-4 space-y-4">
   549→        {/* Exercise selector — scrollable row */}
   550→        <div className="flex gap-1.5 overflow-x-auto pb-1 -mx-1 px-1" style={{ scrollbarWidth: 'none' }}>
   551→          {exercises.map((ex) => (
   552→            <button
   553→              key={ex.id}
   554→              onClick={() => handleExerciseChange(ex.id)}
   555→              className={cn(
   556→                'flex-shrink-0 flex items-center justify-center gap-1 h-8 rounded-lg text-[11px] font-medium transition-all duration-200 px-2.5',
   557→                activeExercise === ex.id
   558→                  ? 'bg-gradient-to-r from-orange-500 to-red-500 text-white shadow-md shadow-orange-500/25'
   559→                  : 'bg-muted/50 text-muted-foreground hover:bg-muted',
   560→              )}
   561→            >
   562→              {ex.icon}
   563→              <span>{ex.name}</span>
   564→            </button>
   565→          ))}
   566→        </div>
   567→
   568→        {/* Exercise description (collapsible) */}
   569→        <div className="overflow-hidden">
   570→          <button
   571→            onClick={() => setShowDescription((p) => !p)}
   572→            className="flex items-center gap-1.5 text-[11px] text-muted-foreground hover:text-foreground transition-colors w-full"
   573→          >
   574→            <Info className="h-3.5 w-3.5" />
   575→            <span className="font-medium">راهنمای تمرین</span>
   576→            {showDescription ? <ChevronUp className="h-3 w-3 mr-auto" /> : <ChevronDown className="h-3 w-3 mr-auto" />}
   577→          </button>
   578→          <AnimatePresence>
   579→            {showDescription && (
   580→              <motion.div
   581→                initial={{ height: 0, opacity: 0 }}
   582→                animate={{ height: 'auto', opacity: 1 }}
   583→                exit={{ height: 0, opacity: 0 }}
   584→                transition={{ duration: 0.2 }}
   585→                className="overflow-hidden"
   586→              >
   587→                <p className="text-[11px] leading-relaxed text-muted-foreground pt-2 pr-1">
   588→                  {currentExerciseDef.description}
   589→                </p>
   590→              </motion.div>
   591→            )}
   592→          </AnimatePresence>
   593→        </div>
   594→
   595→        {/* Speed control for exercises that support it */}
   596→        {currentExerciseDef.hasSpeedControl && (
   597→          <div className="flex items-center justify-center gap-2">
   598→            <Gauge className="h-3.5 w-3.5 text-muted-foreground" />
   599→            <span className="text-[11px] text-muted-foreground">سرعت:</span>
   600→            {(Object.keys(SPEED_CONFIG) as SpeedId[]).map((s) => (
   601→              <button
   602→                key={s}
   603→                onClick={() => setSpeed(s)}
   604→                className={cn(
   605→                  'h-7 px-2.5 rounded-md text-[11px] font-medium transition-all',
   606→                  speed === s
   607→                    ? 'bg-primary text-primary-foreground shadow-sm'
   608→                    : 'bg-muted/50 text-muted-foreground hover:bg-muted',
   609→                )}
   610→              >
   611→                {SPEED_CONFIG[s].label}
   612→              </button>
   613→            ))}
   614→          </div>
   615→        )}
   616→
   617→        {/* Current note display — circle stays, only text changes */}
   618→        <div className="flex justify-center py-2">
   619→          <div
   620→            className={cn(
   621→              'relative h-32 w-32 rounded-full flex flex-col items-center justify-center transition-all duration-150',
   622→              'bg-gradient-to-br from-muted/80 to-muted/40 border-2',
   623→              currentStepData && isPlaying
   624→                ? 'border-orange-400 shadow-lg shadow-orange-500/20'
   625→                : 'border-border/30 hover:border-border/50',
   626→            )}
   627→          >
   628→            {/* Pulsing ring when playing */}
   629→            {currentStepData && isPlaying && (
   630→              <motion.div
   631→                className="absolute inset-0 rounded-full border-2 border-orange-400/40"
   632→                animate={{ scale: [1, 1.15, 1], opacity: [0.4, 0, 0.4] }}
   633→                transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   634→              />
   635→            )}
   636→
   637→            {currentStepData && isPlaying ? (
   638→              <>
   639→                {/* Note name — instant swap, no exit/enter animation */}
   640→                <span
   641→                  key={`n-${currentStep}`}
   642→                  className={cn(
   643→                    'text-3xl font-bold bg-gradient-to-br bg-clip-text text-transparent',
   644→                    NOTE_GRADIENTS[currentStepData.note] ?? 'from-gray-400 to-gray-500',
   645→                  )}
   646→                >
   647→                  {currentStepData.solfege}
   648→                </span>
   649→                <span
   650→                  key={`s-${currentStep}`}
   651→                  className="text-xs text-muted-foreground mt-0.5"
   652→                >
   653→                  {currentStepData.note}{toPersianNum(currentStepData.octave)}
   654→                </span>
   655→              </>
   656→            ) : (
   657→              <>
   658→                <span className="text-2xl text-muted-foreground/40">
   659→                  {currentExerciseDef.icon}
   660→                </span>
   661→                <span className="text-[11px] text-muted-foreground mt-1">
   662→                  {currentExerciseDef.name}
   663→                </span>
   664→              </>
   665→            )}
   666→          </div>
   667→        </div>
   668→
   669→        {/* Step progress dots */}
   670→        <div className={cn(
   671→          'flex items-center justify-center flex-wrap',
   672→          steps.length > 15 ? 'gap-1' : 'gap-1.5',
   673→        )}>
   674→          {steps.map((step, i) => {
   675→            const isCurrent = i === currentStep && isPlaying;
   676→            const isPast = i < currentStep && isPlaying;
   677→            return (
   678→              <motion.div
   679→                key={i}
   680→                initial={false}
   681→                animate={{
   682→                  scale: isCurrent ? 1.4 : 1,
   683→                  opacity: isPast ? 0.4 : isCurrent ? 1 : 0.25,
   684→                }}
   685→                transition={{ type: 'spring', stiffness: 400, damping: 25 }}
   686→                className={cn(
   687→                  'rounded-full transition-colors duration-200',
   688→                  steps.length > 15 ? 'h-1.5 w-1.5' : 'h-2 w-2',
   689→                  isCurrent
   690→                    ? cn('bg-gradient-to-br', NOTE_GRADIENTS[step.note] ?? 'from-gray-400 to-gray-500')
   691→                    : isPast
   692→                      ? 'bg-muted-foreground/40'
   693→                      : 'bg-border',
   694→                )}
   695→              />
   696→            );
   697→          })}
   698→        </div>
   699→
   700→        {/* Play / Stop button */}
   701→        <div className="flex justify-center">
   702→          <Button
   703→            onClick={handlePlay}
   704→            className={cn(
   705→              'h-11 px-8 rounded-full font-medium transition-all duration-300',
   706→              isPlaying
   707→                ? 'bg-gradient-to-r from-red-500 to-rose-600 hover:from-red-600 hover:to-rose-700 shadow-lg shadow-red-500/25'
   708→                : 'bg-gradient-to-r from-orange-500 to-red-500 hover:from-orange-600 hover:to-red-600 shadow-lg shadow-orange-500/25',
   709→            )}
   710→          >
   711→            <AnimatePresence mode="wait" initial={false}>
   712→              {isPlaying ? (
   713→                <motion.span
   714→                  key="stop"
   715→                  initial={{ scale: 0, rotate: -90 }}
   716→                  animate={{ scale: 1, rotate: 0 }}
   717→                  exit={{ scale: 0, rotate: 90 }}
   718→                  className="flex items-center gap-2"
   719→                >
   720→                  <Square className="h-4 w-4" fill="currentColor" />
   721→                  توقف
   722→                </motion.span>
   723→              ) : (
   724→                <motion.span
   725→                  key="play"
   726→                  initial={{ scale: 0, rotate: 90 }}
   727→                  animate={{ scale: 1, rotate: 0 }}
   728→                  exit={{ scale: 0, rotate: -90 }}
   729→                  className="flex items-center gap-2"
   730→                >
   731→                  <Play className="h-4 w-4" fill="currentColor" />
   732→                  شروع
   733→                </motion.span>
   734→              )}
   735→            </AnimatePresence>
   736→          </Button>
   737→        </div>
   738→
   739→        {/* Step counter */}
   740→        {isPlaying && currentStep >= 0 && (
   741→          <motion.p
   742→            initial={{ opacity: 0 }}
   743→            animate={{ opacity: 1 }}
   744→            className="text-center text-[11px] text-muted-foreground"
   745→          >
   746→            نت {toPersianNum(currentStep + 1)} از {toPersianNum(steps.length)}
   747→          </motion.p>
   748→        )}
   749→      </CardContent>
   750→    </Card>
   751→  );
   752→}
