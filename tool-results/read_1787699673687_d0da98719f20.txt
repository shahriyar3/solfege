    50→function toPersianNum(n: number): string {
    51→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    52→  return String(n).replace(/\d/g, (d) => persianDigits[parseInt(d)]);
    53→}
