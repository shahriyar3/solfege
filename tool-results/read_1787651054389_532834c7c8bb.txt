    45→function toPersianNum(n: number): string {
    46→  const persianDigits = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    47→  return String(n).replace(/\d/g, (d) => persianDigits[parseInt(d)]);
    48→}
